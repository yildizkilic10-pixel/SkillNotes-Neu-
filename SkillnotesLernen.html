      }


      box.innerHTML = `<h4>${title}</h4><p>${text}</p>`;

      box.scrollIntoView({ behavior: 'smooth', block: 'start' });

    }


    function filterMaterials() {

      const query = document.getElementById('filterInput').value.toLowerCase().trim();

      const items = document.querySelectorAll('#materialList .material-item:not(#selectedMaterial)');


      items.forEach(item => {

        const text = (item.dataset.keywords || '') + ' ' + item.innerText.toLowerCase();

        item.style.display = text.includes(query) ? '' : 'none';

      });

    }


    function sendMessage() {

      const input = document.getElementById('chatInput');

      const text = input.value.trim();

      if (!text) return;


      const msg = document.createElement('div');

      msg.className = 'msg me';

      msg.textContent = text;

      document.getElementById('messages').appendChild(msg);

      input.value = '';

      msg.scrollIntoView({ behavior: 'smooth' });

    }


    document.getElementById('filterInput').addEventListener('input', filterMaterials);

    document.getElementById('chatInput').addEventListener('keydown', function(event){

      if(event.key === 'Enter') sendMessage();

    });
